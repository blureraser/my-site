本题是二维前缀和模板题。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m, c;
    cin >> n >> m >> c;
    vector<vector<ll>> sum(m + 1, vector<ll>(n + 1, 0));
    for(int i = 1; i <= n; i++) {
        for(int j = 1; j <= m; j++) {
            ll t; cin >> t;
            sum[i][j] = sum[i - 1][j] + sum[i][j - 1] - sum[i - 1][j - 1] + t;
        }
    }

    ll maxi = -1;
    int ansx, ansy;
    for(int i = c; i <= n; i++) {
        for(int j = c; j <= m; j++) {
            ll t = sum[i][j] - sum[i - c][j] - sum[i][j - c] + sum[i - c][j - c];
            if(maxi <= t) {
                maxi = t;
                ansx = i - c + 1; ansy = j - c + 1;
            }
        }
    }
    cout << ansx << " "<< ansy;

    return 0;
}
```
