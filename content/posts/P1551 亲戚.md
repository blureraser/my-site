#include <bits/stdc++.h>
using namespace std;
using ll = long long;

vector<vector<int>> g;
void dfs(int u, vector<int> &got, int iden) {
    if(got[u]) return;
    got[u] = iden;
    for(int v : g[u]) {
        dfs(v, got, iden);
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m, p;
    cin >> n >> m >> p;
    g.assign(n + 1, vector<int>(0));
    for(int i = 0; i < m; i++) {
        int a, b; cin >> a >> b;
        g[a].push_back(b);
        g[b].push_back(a);
    }

    vector<int> got(n + 1, 0);
    for(int i = 1; i <= n; i++) {
        if(got[i]) continue;
        dfs(i, got, i);
    }
    
    for(int i = 0; i < p; i++) {
        int a, b; cin >> a >> b;
        cout << ((got[a] == got[b]) ? "Yes" : "No") << "\n";
    }

    return 0;
}
