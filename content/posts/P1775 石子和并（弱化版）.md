区间 DP非常好理解，`dp[i][j]` 表示区间i到j的最优值。唯一要注意的是dp更新时的依赖关系，在本题中区间长的dp结果总是依赖更短的dp，所以外层循环是区间长度（上界下界偏移量）并从小到大循环。

P1775：
```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;
    vector<int> m(n + 1, 0);
    vector<vector<int>> ms(n + 1, vector<int>(n + 1, 0));
    vector<vector<int>> dp(n + 1, vector<int>(n +  1, INT_MAX));
    for(int i = 0; i <= n; i++) dp[i][i] = 0;
    for(int i = 1; i <= n; i++) {
        int tmp; cin >> tmp;
        m[i] = m[i - 1] + tmp;
    }

    for(int l = 0; l < n; l++){
        for(int i = l + 1; i <= n; i++) {
            for(int k = i - l; k < i; k++) {
                dp[i - l][i] = min(dp[i - l][i], dp[i - l][k] + dp[k + 1][i] + m[i] - m[i - l - 1]);
            }
        }
    }
    cout << dp[1][n];

    return 0;
}
```

P1880 只是将石子排列变成了环形，我的第一想法是直接把环形当长度2n的队列来做：
```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;
    vector<int> a(n * 2 + 1, 0);
    vector<int> sum(n * 2 + 1, 0);
    for(int i = 1; i <= n; i++) {
        int tmp; cin >> tmp;
        a[i] = tmp; a[i + n] = tmp;
    }
    for(int i = 1; i <= 2 * n; i++) sum[i] = sum[i - 1] + a[i];

    vector<vector<int>> dpM(2 * n + 1, vector<int>(2 * n + 1, INT_MIN));
    vector<vector<int>> dp(2 * n + 1, vector<int>(2 * n + 1, INT_MAX));
    for(int i = 0; i <= 2*n; i++) {
        dp[i][i] = 0;
        dpM[i][i] = 0;
    }
    for(int l = 0; l < n; l++) {
        for(int i = l; i <= 2 * n; i++) {
            for(int j = i - l; j < i; j++) {
                dp[i - l][i] = min(dp[i - l][i], dp[i - l][j] + dp[j + 1][i] + sum[i] - sum[i - l - 1]);
                dpM[i - l][i] = max(dpM[i - l][i], dpM[i - l][j] + dpM[j + 1][i] + sum[i] - sum[i - l - 1]);
            }
        }
    }
    int res = INT_MAX;
    int resM = INT_MIN;
    for(int i = 1; i <= n + 1; i++) {
        res = min(res, dp[i][i + n - 1]);
        resM = max(resM, dpM[i][i + n - 1]);
    }

    cout << res << "\n";
    cout << resM << "\n";
    return 0;
}
```
可以通过控制循环范围不重复计算来进一步减小复杂度的常数。
