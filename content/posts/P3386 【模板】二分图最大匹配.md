二分图最大匹配可以作为常见的岗位分配问题的抽象。核心数学方法是增广路定理，程序本质也是如何寻找增广路。

模板使用的返回布尔值的递归方法实在是优雅。
```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

vector<vector<int>> g;
vector<int> matchR;
vector<bool> visitedR;

bool tryAugment(int u) {
    for(int v : g[u]) {
        if(visitedR[v]) continue;
        visitedR[v] = true;

        if(matchR[v] == 0 || tryAugment(matchR[v])) {
            matchR[v] = u;
            return true;
        }
    }
    return false;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m, e;
    cin >> n >> m >> e;
    g.assign(n + 1, vector<int>(0, 0));
    matchR.assign(m + 1, 0);
    for(int i = 0; i < e; i++) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
    }

    int maxM = 0;
    for(int i = 1; i <= n; i++) {
        visitedR.assign(m + 1, false);
        if(tryAugment(i)) maxM++;
    }
    cout << maxM;

    return 0;
}
```
