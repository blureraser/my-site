单调队列模板题。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, k;
    cin >> n >> k;
    vector<int> a(n, 0);
    for(int i = 0; i < n; i++) cin >> a[i];

    deque<int> dq;
    for(int i = 0; i < n; i++) {
        while(!dq.empty()) {
            if(a[dq.back()] < a[i]) dq.pop_back();
            else break;
        }
        dq.push_back(i);
        if(dq.front() <= i - k) dq.pop_front();
        if(i >= k - 1) cout << a[dq.front()] << "\n";
    }


    return 0;
}
```
