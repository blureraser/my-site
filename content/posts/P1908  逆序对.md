此处用到的算法可以认为是在归并排序基础上的扩展。

核心是利用分段后的逆序对数量递推关系：

>整段逆序对数 = 分段1逆序对数 + 分段2逆序对数 + 两端间的逆序对数

同时第三项可以通过归并排序的分段均已排序好的特性加速，使得复杂度仅为排序本身的$O(n logn)$。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

vector<int> a;
vector<int> tmp;

ll merge(int begin, int end) {
    if(begin == end) return 0;
    int mid = (begin + end) / 2;
    ll left = merge(begin, mid);
    ll right = merge(mid + 1, end);

    ll add = 0;
    int i = begin, j = mid + 1, k = begin;
    while(i <= mid && j <= end) {
        if(a[i] <= a[j]) {
            tmp[k] = a[i];
            k++; i++;
        }
        else {
            add += mid - i + 1;
            tmp[k] = a[j];
            k++; j++;
        }
    }
    for(; i <= mid; k++, i++) tmp[k] = a[i];
    for(; j <= end; k++, j++) tmp[k] = a[j];
    for(int l = begin; l <= end; l++) a[l] = tmp[l];

    return add + left + right;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;

    a.assign(n, 0);
    tmp.assign(n, 0);
    for(int i = 0; i < n; i++) {
        cin >> a[i];
    }
    cout << merge(0, a.size() - 1);

    return 0;
}
```
